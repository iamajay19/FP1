        for (class,max,header) in [(Class::Small,16384,3),(Class::Bundle,32768,3),(Class::Crl,16777216,5)] {
            for delta in [-1isize,0,1] {
                let n=(max as isize+delta) as usize;
                let payload=vec![0;n-header];
                let mut b=if header==3 {let mut v=vec![0x59];v.extend_from_slice(&((n-header)as u16).to_be_bytes());v}
                    else {let mut v=vec![0x5a];v.extend_from_slice(&((n-header)as u32).to_be_bytes());v};

                else {assert!(matches!(validated::<Blob>(&b,class),Err(Error::SizeLimit)));assert!(matches!(deterministic(&Blob(&payload),class),Err(Error::SizeLimit)));}
            }
        }
    }
